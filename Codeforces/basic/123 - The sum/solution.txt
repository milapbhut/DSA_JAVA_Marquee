import java.util.*;

public class Main {
    public static void main(String[] args) {
      Scanner sc = new Scanner(System.in);
      int n = sc.nextInt();
      long f1 = 1;
      long f2 = 1;
      long fibonacciSum = 1;  
        for (int i = 0; i < n - 1; i++) {
            long temp = f1 + f2;
            f1 = f2;
            f2 = temp;
            fibonacciSum += f1;
        }
      System.out.println(fibonacciSum);
    }
}
